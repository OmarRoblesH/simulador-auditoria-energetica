// Agregar un nuevo registro manual en la tabla de Equipos function agregarFilaEquipo(reg = '', nombre = '', categoria = 'Otros aparatos eléctricos', consumo = 0) { const tbody = document.getElementById('tbodyEquipos'); const codigoUnico = reg || 'EQ-' + Math.floor(100000 + Math.random() * 900000); const nuevaFila = document.createElement('tr'); nuevaFila.innerHTML = `
${categoria === 'Sistemas de iluminación' ? 'W' : 'kW'}
`; tbody.appendChild(nuevaFila); } // Ajuste automático de unidades según la categoría del equipo function actualizarUnidadEquipo(selectElem) { const fila = selectElem.closest('tr'); const spanUnidad = fila.querySelector('.span-unidad-equipo'); if (selectElem.value === 'Sistemas de iluminación') { spanUnidad.textContent = 'W'; } else { spanUnidad.textContent = 'kW'; } } // Eliminar fila y actualizar gráficos dinámicamente function eliminarFilaEquipo(btn) { const fila = btn.closest('tr'); if (fila) { fila.remove(); if (document.getElementById('step-5').classList.contains('active')) { renderCharts(); } } } // Procesamiento OCR inteligente de etiquetas y equipos eléctricos function ejecutarOCREquipo(fileInput) { const file = fileInput.files[0]; if (!file) return; document.getElementById('ocrEquipoSpinner').classList.remove('d-none'); Tesseract.recognize( file, 'spa+eng', { logger: m => console.log(m) } ).then(({ data: { text } }) => { document.getElementById('ocrEquipoSpinner').classList.add('d-none'); const textoLimpio = text.toLowerCase(); let categoriaDetectada = 'Otros aparatos eléctricos'; let nombreDetectado = 'Aparato Eléctrico Detectado'; let consumoDetectado = 1.5; // Reglas de coincidencia por palabras clave en la etiqueta/foto if (textoLimpio.includes('led') || textoLimpio.includes('lamp') || textoLimpio.includes('foco') || textoLimpio.includes('lumens')) { categoriaDetectada = 'Sistemas de iluminación'; nombreDetectado = 'Lámpara / Foco LED'; consumoDetectado = 50; } else if (textoLimpio.includes('split') || textoLimpio.includes('aires') || textoLimpio.includes('btu') || textoLimpio.includes('r410a') || textoLimpio.includes('refrigerador')) { categoriaDetectada = 'Generación de frío y climatización'; nombreDetectado = 'Aire Acondicionado / Refrigeración'; consumoDetectado = 2.2; } else if (textoLimpio.includes('boiler') || textoLimpio.includes('calentador') || textoLimpio.includes('horno') || textoLimpio.includes('caldera')) { categoriaDetectada = 'Generación de calor'; nombreDetectado = 'Sistema de Calefacción / Caldera'; consumoDetectado = 4.5; } else if (textoLimpio.includes('hp') || textoLimpio.includes('motor') || textoLimpio.includes('bomba') || textoLimpio.includes('rpm') || textoLimpio.includes('kw')) { categoriaDetectada = 'Fuerza y motores'; nombreDetectado = 'Motor Eléctrico / Bomba'; consumoDetectado = 5.5; } // Extracción simple de valores numéricos acompañados de W o kW si se encuentran en el texto const matchKW = text.match(/(\d+[\.,]?\d*)\s*(kw|kw/i); const matchW = text.match(/(\d+[\.,]?\d*)\s*(w|watts)/i); if (matchKW) { consumoDetectado = parseFloat(matchKW[1].replace(',', '.')); } else if (matchW && categoriaDetectada === 'Sistemas de iluminación') { consumoDetectado = parseFloat(matchW[1].replace(',', '.')); } const registroGenerado = 'OCR-' + Math.floor(1000 + Math.random() * 9000); agregarFilaEquipo(registroGenerado, nombreDetectado, categoriaDetectada, consumoDetectado); alert(`¡Equipo escaneado con éxito!\nDetectado: ${nombreDetectado}\nCategoría: ${categoriaDetectada}\nPotencia/Consumo: ${consumoDetectado}`); }).catch(err => { document.getElementById('ocrEquipoSpinner').classList.add('d-none'); alert("No se pudo procesar la imagen del equipo. Verifique la claridad de la toma o agregue el registro manualmente."); }); } // Renderizado y cálculo acumulativo de consumos para los Dashboards del Bloque 5 function renderCharts() { if (chart1) chart1.destroy(); if (chart2) chart2.destroy(); // Sumatoria de potencia/consumos dinámicos registrados en la tabla del Bloque 4 const acumuladoCategorias = { 'Sistemas de iluminación': 0, 'Generación de frío y climatización': 0, 'Generación de calor': 0, 'Fuerza y motores': 0, 'Agua y agua caliente sanitaria': 0, 'Otros aparatos eléctricos': 0 }; const filasEquipos = document.querySelectorAll('#tbodyEquipos tr'); filasEquipos.forEach(fila => { const selectCat = fila.querySelector('select'); const inputConsumo = fila.querySelector('.input-potencia'); if (selectCat && inputConsumo) { const cat = selectCat.value; let val = parseFloat(inputConsumo.value) || 0; // Si la unidad es Watts, se pasa a kW para unificar la escala del Dashboard if (cat === 'Sistemas de iluminación') val = val / 1000; if (acumuladoCategorias[cat] !== undefined) { acumuladoCategorias[cat] += val; } else { acumuladoCategorias[cat] = val; } } }); // Gráfica de línea const ctx1 = document.getElementById('chartConsumoMensual').getContext('2d'); chart1 = new Chart(ctx1, { type: 'line', data: { labels: ['Ene', 'Feb', 'Mar', 'Abr', 'May', 'Jun', 'Jul', 'Ago', 'Sep', 'Oct', 'Nov', 'Dic'], datasets: [{ label: 'Consumo Estimado (kWh)', data: [1100, 1900, 3800, 2100, 2400, 1800, 2900, 4800, 2200, 2100, 1800, 2300], backgroundColor: 'rgba(37, 99, 235, 0.2)', borderColor: '#2563eb', fill: true, tension: 0.2 }] }, options: { responsive: true, plugins: { legend: { display: true } } } }); // Gráfica de barras alimentada directamente del inventario de equipos const ctx2 = document.getElementById('chartConsumoCategoria').getContext('2d'); chart2 = new Chart(ctx2, { type: 'bar', data: { labels: Object.keys(acumuladoCategorias), datasets: [{ label: 'Potencia Total Instalada (kW)', data: Object.values(acumuladoCategorias), backgroundColor: [ '#3b82f6', '#06b6d4', '#ef4444', '#f59e0b', '#10b981', '#6b7280' ] }] }, options: { responsive: true, plugins: { legend: { display: false }, title: { display: true, text: 'Potencia Total por Categoría (kW)' } } } });
